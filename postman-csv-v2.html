`; /** * Constructs the visualizer payload from Postman's response * Includes error handling and support for various JSON structures */ function constructVisualizerPayload(pm) { try { // Default in case of errors let data = []; let json = null; // Try to parse response try { json = pm.response.json(); } catch (e) { console.error('Error parsing JSON response:', e); // Check if it's an array-like string that just needs parsing try { const responseText = pm.response.text(); if (responseText && (responseText.startsWith('[') || responseText.startsWith('{'))) { json = JSON.parse(responseText); } } catch (innerError) { console.error('Could not recover from JSON parsing error:', innerError); } } // Handle different response structures if (Array.isArray(json)) { data = json; } else if (json && typeof json === 'object') { // Look for array properties data = null; // First, look for properties with large arrays - they're likely the main data let largestArraySize = 0; let largestArrayKey = null; for (const key in json) { if (Array.isArray(json[key]) && json[key].length > largestArraySize) { largestArraySize = json[key].length; largestArrayKey = key; } } if (largestArrayKey && largestArraySize > 0) { data = json[largestArrayKey]; } else { // If no arrays or all empty, look for any array for (const key in json) { if (Array.isArray(json[key])) { data = json[key]; break; } } } // If still no valid array found, use the object itself if (!data) data = [json]; } else { // For primitive responses or invalid/empty responses data = [{value: String(json)}]; } // Ensure we don't have too many records (performance) if (data.length > 1000) { console.warn(`Large dataset detected (${data.length} records). Limiting to first 1000 for performance.`); data = data.slice(0, 1000); } // Flatten nested objects for table display const flat = data.map(obj => { const res = {}; // Recursive function to flatten object (function flatten(o, prefix = '') { for (const key in o) { if (!o.hasOwnProperty(key)) continue; const value = o[key]; const newKey = prefix ? `${prefix}.${key}` : key; if (Array.isArray(value)) { // Handle arrays by converting to string but in a readable format res[newKey] = JSON.stringify(value) .replace(/,/g, ', ') .replace(/\[/g, '[') .replace(/\]/g, ']') .replace(/"/g, ''); } else if (value && typeof value === 'object' && value !== null) { flatten(value, newKey); } else { // Handle null, undefined, and primitive values res[newKey] = value === null || value === undefined ? '' : String(value); } } })(obj); return res; }); // Extract unique headers across all objects const headers = [...new Set(flat.flatMap(r => Object.keys(r)))]; // Create rows from flattened data const rows = flat.map(r => headers.map(h => r[h] || '')); return { headers, rows }; } catch (e) { console.error('Fatal error in constructVisualizerPayload:', e); // Return minimal payload that won't break the visualizer return { headers: ['Error'], rows: [['Failed to process response. See console for details.']] }; } } // Set the visualizer with proper error handling try { pm.visualizer.set(template, constructVisualizerPayload(pm)); } catch (e) { console.error('Error setting visualizer:', e); pm.visualizer.set(`

Check the console for more details.

`); }
